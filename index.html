WeaponSystem.lua 
local Weapon = {}
Weapon.__index = Weapon

function Weapon.new(config)
  local self = setmetatable({}, Weapon)
  self.damage   = config.damage   or 25
  self.cooldown = config.cooldown or 0.5
  self._lastFired = 0
  return self
end

function Weapon:canFire()
  return tick() - self._lastFired
       >= self.cooldown
end

return Weapon
